# ==============================================================================
# Dylan Ferreira - Enterprise Production Delta Lake Ingestion Routine
# Pattern: Medallion Architecture Bronze -> Silver Conformed Merge
# Invariants: Idempotent merge, SHA-256 change detection, Zero duplicate rows
# ==============================================================================

from pyspark.sql import functions as F
from delta.tables import DeltaTable

def execute_silver_conformed_merge(
    spark_session,
    bronze_df, 
    silver_table_path: str, 
    primary_keys: list, 
    exclude_hash_cols: list
):
    # 1. Compute deterministic SHA-256 row hash for high-throughput CDC
    hash_columns = [
        F.coalesce(F.col(c).cast("string"), F.lit("__NULL_TOKEN__"))
        for c in bronze_df.columns 
        if c not in exclude_hash_cols
    ]
    
    staged_updates_df = (
        bronze_df
        .withColumn("_row_hash_sha256", F.sha2(F.concat_ws("||", *hash_columns), 256))
        .withColumn("_silver_conformed_ts", F.current_timestamp())
    )

    # 2. Bind conformed target Delta table instance
    target_silver_table = DeltaTable.forPath(spark_session, silver_table_path)
    
    # 3. Build compound primary key join predicate
    merge_condition = " AND ".join([
        f"target.{k} = source.{k}" for k in primary_keys
    ])
    
    # 4. Execute atomic Delta Merge: update ONLY on detected byte-level divergence
    (
        target_silver_table.alias("target")
        .merge(
            source=staged_updates_df.alias("source"),
            condition=merge_condition
        )
        .whenMatchedUpdate(
            condition="target._row_hash_sha256 != source._row_hash_sha256",
            set={"*": "source.*"}
        )
        .whenNotMatchedInsertAll()
        .execute()
    )

    # 5. Delta maintenance & Z-Order clustering for sub-second analytical scans
    spark_session.sql(f"OPTIMIZE delta.`{silver_table_path}` ZORDER BY ({','.join(primary_keys)})")
    return True
